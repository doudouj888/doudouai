    function resolveApiEndpoints() {
      const endpoints = [];
      const queryApi = new URLSearchParams(window.location.search).get("api");
      if (queryApi) endpoints.push(queryApi);

      if (Array.isArray(window.API_ENDPOINTS)) {
        endpoints.push(...window.API_ENDPOINTS);
      }

      endpoints.push(DEFAULT_API);
      return [...new Set(
        endpoints
          .map((v) => String(v || "").trim())
          .filter((v) => /^https?:\/\//i.test(v))

    async function fetchWithTimeout(url, options = {}, timeoutMs = REQUEST_TIMEOUT_MS) {
      const controller = new AbortController();
      const timer = setTimeout(() => controller.abort(), timeoutMs);
      try {
        return await fetch(url, { ...options, signal: controller.signal });
      } finally {
        clearTimeout(timer);
      }
    }

    async function submitWithFallback(payload) {
      let lastError = "";

      for (const endpoint of API_ENDPOINTS) {
        try {
          const res = await fetchWithTimeout(endpoint, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify(payload),
          });

          const text = await res.text();
          let data = {};
          try { data = JSON.parse(text); } catch { data = { message: text }; }

          if (res.ok) {
            return { ok: true, data, endpoint };
          }

          const retryable = res.status === 404 || res.status === 408 || res.status === 429 || res.status >= 500;
          if (!retryable) {
            return { ok: false, data, endpoint, status: res.status };
          }

          lastError = `接口 ${endpointLabel(endpoint)} 返回 ${res.status}`;
        } catch (e) {
          if (e?.name === "AbortError") {
            lastError = `连接 ${endpointLabel(endpoint)} 超时`;
          } else {
            lastError = `无法连接 ${endpointLabel(endpoint)}`;
          }
        }
      }

      throw new Error(lastError || "所有接口不可用");
    }

    /* ================== Modal ================== */
    let modalShown = false;

    function showModal(type, title, msg, options = {}) {
      if (!modalEl || !modalCardEl || !modalBackdropEl || !modalTitleEl || !modalMsgEl) {
        alert(`${title || "提示"}\n${msg || ""}`);
        if (typeof options.onClose === "function") options.onClose();
        return;
      }

      const modalType = type || "info";
      modalEl.dataset.type = modalType;
      modalTitleEl.textContent = title || (modalType === "success" ? "成功" : "失败");
      modalMsgEl.textContent = msg || "";

      if (modalIconEl) {
        modalIconEl.textContent = modalType === "success" ? "✓" : modalType === "error" ? "!" : "i";
      }

      modalEl.classList.remove("hidden");
      modalShown = true;

      requestAnimationFrame(() => {
        modalEl.classList.add("show");
      });

      modalEl.__onClose = options.onClose || null;
    }

    function hideModal() {
      if (!modalShown || !modalEl) return;
      modalEl.classList.remove("show");

      setTimeout(() => {
        modalEl.classList.add("hidden");
        modalShown = false;

        if (typeof modalEl.__onClose === "function") {
          try { modalEl.__onClose(); } catch {}
        }
        modalEl.__onClose = null;
      }, 170);
    }

    modalCloseBtnEl?.addEventListener("click", hideModal);
    modalOkBtnEl?.addEventListener("click", hideModal);
    modalBackdropEl?.addEventListener("click", hideModal);
    document.addEventListener("keydown", (e) => {
      if (e.key === "Escape") hideModal();
    });

    /* ================== 表单状态 ================== */
    function refresh() {
      if (!inviteCodeEl || !emailEl || !confirmEl || !submitBtn) return;

      const code = inviteCodeEl.value.trim();
      const em = emailEl.value.trim();
      const ok = code.length > 0 && isValidEmail(em) && confirmEl.checked;

      submitBtn.disabled = !ok;
      submitBtn.classList.toggle("is-enabled", ok);
    }

    ["input", "change"].forEach((evt) => {
      inviteCodeEl?.addEventListener(evt, refresh);
      emailEl?.addEventListener(evt, refresh);
      confirmEl?.addEventListener(evt, refresh);
    });

    /* ================== 提交逻辑 ================== */
    submitBtn?.addEventListener("click", async () => {
      if (!inviteCodeEl || !emailEl || !confirmEl || !submitBtn) return;
      if (submitBtn.disabled) return;

      const oldText = submitBtn.textContent;
      submitBtn.disabled = true;
      submitBtn.textContent = "提交中...";
      submitBtn.classList.remove("is-enabled");

      try {
        setDebug("正在提交，请稍候...", "muted");

        const payload = {
          inviteCode: inviteCodeEl.value.trim(),
          email: emailEl.value.trim(),
        };
        const result = await submitWithFallback(payload);

        if (!result.ok) {
          showModal("error", "兑换失败", result.data?.message || "请求失败");
          setDebug(`业务拒绝：${result.data?.message || `HTTP ${result.status}`}`, "error");
          return;
        }

        showModal("success", "兑换成功", result.data?.message || "操作成功", {
          onClose: () => {
            inviteCodeEl.value = "";
            emailEl.value = "";
            confirmEl.checked = false;
            refresh();
          },
        });
        setDebug(`提交成功（通道：${endpointLabel(result.endpoint)}）`, "success");
      } catch (e) {
        const msg = "当前网络无法连接兑换接口。请稍后重试，或切换网络后再提交。";
        showModal("error", "网络错误", msg);
        setDebug(`${e?.message || "网络请求失败"}。${msg}`, "error");
      } finally {
        submitBtn.textContent = oldText;
        refresh();
      }
    });

    /* ================== 绑定按钮 ================== */
    window.addEventListener("DOMContentLoaded", () => {
      toggleGuaranteeBtn?.addEventListener("click", () => {
        if (!guaranteeEl) return;
        const willShow = guaranteeEl.classList.contains("hidden");
        guaranteeEl.classList.toggle("hidden");
        if (arrowEl) arrowEl.textContent = willShow ? "▴" : "▾";
      });
    });

    /* ================== 初始化 ================== */
    refresh();
    if (API_ENDPOINTS.length === 1 && API_ENDPOINTS[0].includes("workers.dev")) {
      setDebug("当前仅使用 workers.dev 通道，中国大陆网络可能不稳定。建议绑定自定义域名 API。", "warn");
    }
  </script>
